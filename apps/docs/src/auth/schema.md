# Auth Schema

Sedim's auth module auto-generates database schema for the features you select.

## Drizzle ORM

When using Drizzle, import the schema from `@sedim/auth/schema`:

```typescript
import { authSchema } from '@sedim/auth/schema';
// or per-feature:
import { users, sessions, oauthAccounts } from '@sedim/auth/schema';

// In your drizzle.config.ts:
export default {
  schema: {
    ...authSchema,
    // your other tables
  },
};
```

## Prisma

Prisma schema is generated by `sedim add auth` and injected into `schema.prisma` using Prisma's multi-file schema support.

The generated models are:

```prisma
model users {
  id            String    @id @default(cuid())
  email         String    @unique
  passwordHash  String?
  emailVerified Boolean   @default(false)
  createdAt     DateTime  @default(now())
  updatedAt     DateTime  @updatedAt

  sessions       sessions[]
  oauthAccounts  oauth_accounts[]
  totpCredential totp_credentials?
  backupCodes    backup_codes[]
  otpTokens      otp_tokens[]
  refreshTokens  refresh_tokens[]
}

model sessions {
  id        String   @id @default(cuid())
  userId    String
  expiresAt DateTime
  createdAt DateTime @default(now())
  lastSeenAt DateTime @default(now())

  user users @relation(fields: [userId], references: [id], onDelete: Cascade)
}

model oauth_accounts {
  id           String @id @default(cuid())
  userId       String
  provider     String
  providerUserId String

  user users @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([provider, providerUserId])
}

model totp_credentials {
  id        String   @id @default(cuid())
  userId    String   @unique
  secret    String   // AES-256-GCM encrypted
  salt      String
  verified  Boolean  @default(false)
  createdAt DateTime @default(now())

  user users @relation(fields: [userId], references: [id], onDelete: Cascade)
}

model backup_codes {
  id       String @id @default(cuid())
  userId   String
  code     String // hashed
  usedAt   DateTime?

  user users @relation(fields: [userId], references: [id], onDelete: Cascade)
}

model otp_tokens {
  id        String   @id @default(cuid())
  userId    String
  token     String   // hashed
  type      String   // magic_link | password_reset
  expiresAt DateTime
  usedAt    DateTime?

  user users @relation(fields: [userId], references: [id], onDelete: Cascade)
}

model refresh_tokens {
  id        String   @id @default(cuid())
  userId    String
  token     String   // hashed
  expiresAt DateTime
  createdAt DateTime @default(now())

  user users @relation(fields: [userId], references: [id], onDelete: Cascade)
}
```

## Migrations

After stamping auth, run your ORM's migration command:

```bash
# Drizzle
drizzle-kit generate
drizzle-kit migrate

# Prisma
npx prisma migrate dev --name add_auth
```
